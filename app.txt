#!/bin/bash

set -e

echo "=== Updating packages ==="
sudo apt update

echo "=== Installing dependencies ==="
sudo apt install -y git maven tomcat9 default-jdk

echo "=== Cloning Spring Petclinic ==="
cd /opt

if [ ! -d "/opt/spring-petclinic" ]; then
    sudo git clone https://github.com/spring-projects/spring-petclinic.git
fi

cd /opt/spring-petclinic

echo "=== Configuring application ==="

sudo tee src/main/resources/application.properties > /dev/null <<EOF
spring.datasource.url=jdbc:mysql://x.x.x.x:3306/petclinic
spring.datasource.username=petclinic
spring.datasource.password=StrongPass!123

memcached.server=x.x.x.x:11211

spring.rabbitmq.host=x.x.x.x
spring.rabbitmq.username=appuser
spring.rabbitmq.password=StrongPass!123
EOF

echo "=== Building application ==="
mvn clean package

echo "=== Deploying WAR ==="
sudo cp target/spring-petclinic-*.war /var/lib/tomcat9/webapps/petclinic.war

echo "=== Restarting Tomcat ==="
sudo systemctl enable tomcat9
sudo systemctl restart tomcat9

echo "=== Testing application ==="
sleep 5
curl http://localhost:8080/petclinic/

echo ""
echo "=== APP SETUP COMPLETED ==="
