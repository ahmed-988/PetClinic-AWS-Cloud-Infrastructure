#!/bin/bash
yum update -y


amazon-linux-extras enable java-openjdk17 -y
yum install java-17-openjdk -y


yum install maven git tomcat8 -y
systemctl enable tomcat
systemctl start tomcat


cd /opt
git clone https://github.com/spring-projects/spring-petclinic.git
cd spring-petclinic


sed -i 's|url: jdbc.*|url: jdbc:mysql://10.0.3.10:3306/petclinic|' src/main/resources/application.properties
sed -i 's|host:.*|host: 10.0.4.10|' src/main/resources/application.properties

echo "memcached.server=10.0.5.10" >> src/main/resources/application.properties


mvn clean package


cp target/*.war /usr/share/tomcat/webapps/


systemctl restart tomcat
